const int buzzerPin = 9;
const int fireSensorPin = A1;
const int led1Pin = 11;
const int led2Pin = 3;
const int mq2Pin = A0;
// lowest and highest sensor readings:
const int sensorMin = 0;     //  sensor minimum
const int sensorMax = 1024;  // sensor maximum
void setup() {
Serial.begin(9600);
pinMode(buzzerPin, OUTPUT);
pinMode(fireSensorPin, INPUT);
pinMode(led1Pin, OUTPUT);
pinMode(led2Pin,OUTPUT);
}

void loop(){
int fireValue = analogRead(fireSensorPin);
int smokeValue = analogRead(mq2Pin);
Serial.println(smokeValue);
Serial.println(fireValue);

int range = map(fireValue,  sensorMin, sensorMax, 0, 3);

if (fireValue < 100) {
Serial.println("Fire Detected!");
digitalWrite(buzzerPin, HIGH);
digitalWrite(led1Pin, LOW);
digitalWrite(led2Pin,HIGH);

} else if (smokeValue > 160) {
Serial.println("Smoke Detected!");
Serial.print("Smoke Value: ");
Serial.println(smokeValue);
digitalWrite(buzzerPin, HIGH);
digitalWrite(led1Pin, LOW);
digitalWrite(led2Pin,HIGH);
}
else {
  Serial.println("No Fire or Smoke Detected");
  digitalWrite(buzzerPin, LOW);
digitalWrite(led2Pin, LOW);
digitalWrite(led1Pin,HIGH);
}

delay (1000);
